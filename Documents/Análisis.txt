3.- Solución:

    Se agregó un campo adicional a la tabla portalcfdi.companies para clasificar cada sociedad en qué versión del ERP/SAP está implementada.
    De esta forma facilitará del lado de CanPro cambiar el flujo para las demás entidades, cuando sean migradas a la nueva versión de SAP S4/Hana, siempre y cuando se mantengan exactamente las mismas condiciones de este requerimiento.
    No se agregará esta opción a nivel de interfaz de usuario, debido a que debe ser un cambio controlado. Para el caso de dar de alta una nueva sociedad, se asignará por default ECC.
    Valdría la pena la negociación para dar de alta nuevas entidades/sociedades, por el consumo de las peticiones con el PAC/Connexial.

    ALTER TABLE portalcfdi.companies ADD COLUMN sapversion ENUM('ECC','S4H') NOT NULL DEFAULT 'ECC';

    UPDATE portalcfdi.companies SET sapversion = 'S4H' WHERE crfc = 'CNC820428674';

    Procedimiento Almacenado Actual y Adaptación:

    DELIMITER $$
    CREATE DEFINER = 'portaluser'@'localhost' PROCEDURE cancellations_viewsap()
    BEGIN

        SELECT
            cfdiuuid,
            IF( ISNULL(koneshdata.docDate) OR (LENGTH(TRIM(koneshdata.docDate)) = 0), metadata.issueDate, koneshdata.docDate) AS issueDate,
            IF( ISNULL(koneshdata.docType) OR (LENGTH(TRIM(koneshdata.docType)) = 0), metadata.cfditype, koneshdata.docType) AS cfditype,
            IF( ISNULL(koneshdata.emiRFC) OR (LENGTH(TRIM(koneshdata.emiRFC)) = 0), metadata.rfcEmitter, koneshdata.emiRFC) AS rfcEmitter,
            IF( ISNULL(koneshdata.recRFC) OR (LENGTH(TRIM(koneshdata.recRFC)) = 0), metadata.rfcReceiver, koneshdata.recRFC) AS rfcReceiver,
            IF( ISNULL(koneshdata.folio) OR (LENGTH(TRIM(koneshdata.folio)) = 0), metadata.folio, koneshdata.folio) AS folio,
            IF( ISNULL(koneshdata.serie) OR (LENGTH(TRIM(koneshdata.serie)) = 0), metadata.serie, koneshdata.serie) AS serie,
            IF( koneshdata.total > 0, koneshdata.total, metadata.amount ) AS amount
        FROM
            cancellations
        LEFT JOIN
            koneshdata ON koneshdata.uuid = cfdiuuid
        LEFT JOIN
            metadata ON metadata.uuid = cfdiuuid
        WHERE
            response = 'Pendiente'
        AND
            curstage = 'PAC'
        AND
            cancellations.status IN(1,12);

    END $$
    DELIMITER ;

    DELIMITER $$
    CREATE DEFINER = 'portaluser'@'localhost' PROCEDURE cancellations_viewsap
    (
        IN sapver   VARCHAR(4)
    )
    BEGIN

        SELECT
            c.cfdiuuid,
            k.docDate   AS issueDate,
            k.docType   AS cfditype,
            k.emiRFC    AS rfcEmitter,
            k.recRFC    AS rfcReceiver,
            k.folio,
            k.serie,
            k.total     AS amount
        FROM
            cancellations c
        INNER JOIN
            koneshdata k ON k.uuid = c.cfdiuuid
        WHERE
            c.response = 'Pendiente'
        AND
            c.curstage = 'PAC'
        AND
            c.status IN(1,12)
        AND
            k.recRFC IN(SELECT crfc FROM companies WHERE sapversion = sapver)

        UNION ALL

        SELECT
            c.cfdiuuid,
            m.issueDate,
            m.cfditype,
            m.rfcEmitter,
            m.rfcReceiver,
            m.folio,
            m.serie,
            m.amount
        FROM
            cancellations c
        INNER JOIN
            metadata m ON m.uuid = c.cfdiuuid
        LEFT JOIN
            koneshdata k ON k.uuid = c.cfdiuuid
        WHERE
            c.response = 'Pendiente'
        AND
            c.curstage = 'PAC'
        AND
            c.status IN(1,12)
        AND
            m.rfcReceiver IN(SELECT crfc FROM companies WHERE sapversion = sapver);

    END $$
    DELIMITER ;
